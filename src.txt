===== src/services/productService.js =====
import { db } from '../db/index'

export const productService = {
  // Mapping dari format POS ke format Katalog App
  transformPOSProduct(rawProduct) {
    return {
      id: rawProduct.id,
      name: rawProduct.name,
      category: rawProduct.category || 'UMUM',
      price: rawProduct.price_sell || 0,
      unit: rawProduct.unit || 'PCS',
      stock: rawProduct.qty || 0,
      isAvailable: (rawProduct.qty > 0) && (rawProduct.price_sell > 0),
      // Contoh generator URL Cloudinary berbasis nama/ID produk
      imageUrl: rawProduct.imageUrl || `https://res.cloudinary.com/warung-sinar-pagi/image/upload/v1/products/${rawProduct.id}.jpg`
    }
  },

  // Seed data awal dari JSON POS ke IndexedDB
  async initProductsFromJSON(rawJSONData) {
    const productsCount = await db.products.count()
    if (productsCount === 0 && rawJSONData?.products) {
      const transformedProducts = rawJSONData.products.map(this.transformPOSProduct)
      await db.products.bulkAdd(transformedProducts)
      
      if (rawJSONData?.categories) {
        await db.categories.bulkAdd(rawJSONData.categories)
      }
    }
  },

  async getAllProducts() {
    return await db.products.toArray()
  },

  async getCategories() {
    return await db.categories.toArray()
  }
}

===== src/services/whatsappService.js =====
export const whatsappService = {
  formatOrderMessage(cartItems, customerInfo = {}) {
    const { name = 'Anak Kost', room = '-' } = customerInfo
    
    let text = `*PESANAN BARU - WARUNG SINAR PAGI*\n`
    text += `-----------------------------------\n`
    text += `*Pemesan:* ${name}\n`
    text += `*No. Kost/Kamar:* ${room}\n\n`
    text += `*Rincian Belanjaan:*\n`

    let total = 0
    cartItems.forEach((item, index) => {
      const subtotal = item.price * item.quantity
      total += subtotal
      text += `${index + 1}. ${item.name} (${item.quantity} ${item.unit}) = Rp ${subtotal.toLocaleString('id-ID')}\n`
      if (item.selectedVariant) {
        text += `   _Catatan: ${item.selectedVariant}_\n`
      }
    })

    text += `\n-----------------------------------\n`
    text += `*TOTAL BAYAR:* Rp ${total.toLocaleString('id-ID')}\n\n`
    text += `Mohon konfirmasi pesanan dan info nomor QRIS / COD ya A/Teteh. Terima kasih!`

    return encodeURIComponent(text)
  },

  generateWhatsAppURL(phoneNumber, formattedMessage) {
    const cleanPhone = phoneNumber.replace(/[^0-9]/g, '')
    return `https://wa.me/${cleanPhone}?text=${formattedMessage}`
  }
}

===== src/stores/catalogStore.js =====
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { productService } from '../services/productService'

export const useCatalogStore = defineStore('catalog', () => {
  const products = ref([])
  const categories = ref([])
  const selectedCategory = ref('SEMUA')
  const searchQuery = ref('')
  const isLoading = ref(false)

  // Fetch data dari Dexie.js / IndexedDB
  async function fetchCatalog() {
    isLoading.value = true
    try {
      products.value = await productService.getAllProducts()
      categories.value = await productService.getCategories()
    } catch (error) {
      console.error('Gagal mengambil data katalog:', error)
    } finally {
      isLoading.value = false
    }
  }

  // Filter gabungan berdasarkan pencarian teks dan kategori
  const filteredProducts = computed(() => {
    return products.value.filter(product => {
      const matchCategory = selectedCategory.value === 'SEMUA' || product.category === selectedCategory.value
      const matchSearch = product.name.toLowerCase().includes(searchQuery.value.toLowerCase().trim())
      return matchCategory && matchSearch && product.isAvailable
    })
  })

  function setCategory(categoryName) {
    selectedCategory.value = categoryName
  }

  function setSearchQuery(query) {
    searchQuery.value = query
  }

  return {
    products,
    categories,
    selectedCategory,
    searchQuery,
    isLoading,
    filteredProducts,
    fetchCatalog,
    setCategory,
    setSearchQuery
  }
})

===== src/stores/cartStore.js =====
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { whatsappService } from '../services/whatsappService'

export const useCartStore = defineStore('cart', () => {
  const items = ref([])

  const totalItems = computed(() => items.value.reduce((sum, item) => sum + item.quantity, 0))
  const totalPrice = computed(() => items.value.reduce((sum, item) => sum + (item.price * item.quantity), 0))

  function addItem(product, variant = null) {
    const existing = items.value.find(i => i.id === product.id && i.selectedVariant === variant)
    if (existing) {
      existing.quantity++
    } else {
      items.value.push({
        id: product.id,
        name: product.name,
        price: product.price,
        unit: product.unit,
        selectedVariant: variant,
        quantity: 1
      })
    }
  }

  function removeItem(productId) {
    items.value = items.value.filter(i => i.id !== productId)
  }

  function updateQuantity(productId, qty) {
    const item = items.value.find(i => i.id === productId)
    if (item) {
      item.quantity = qty
      if (item.quantity <= 0) removeItem(productId)
    }
  }

  function checkoutViaWhatsApp(storePhone, customerInfo) {
    const message = whatsappService.formatOrderMessage(items.value, customerInfo)
    const url = whatsappService.generateWhatsAppURL(storePhone, message)
    window.open(url, '_blank')
  }

  return { items, totalItems, totalPrice, addItem, removeItem, updateQuantity, checkoutViaWhatsApp }
})

===== src/composables/useCatalog.js =====
import { onMounted } from 'vue'
import { storeToRefs } from 'pinia'
import { useCatalogStore } from '../stores/catalogStore'
import { productService } from '../services/productService'
import initialPOSData from '../mock/sinar-pagi.json'

export function useCatalog() {
  const catalogStore = useCatalogStore()

  // Ambil state reaktif menggunakan storeToRefs agar reaktivitas Vue tetap terjaga
  const { 
    products, 
    categories, 
    selectedCategory, 
    searchQuery, 
    filteredProducts, 
    isLoading 
  } = storeToRefs(catalogStore)

  // Ambil method/action dari store
  const { setCategory, setSearchQuery, fetchCatalog } = catalogStore

  onMounted(async () => {
    try {
      // 1. Seed data ke Dexie.js dari file mock sinar-pagi.json jika database lokal masih kosong
      await productService.initProductsFromJSON(initialPOSData)

      // 2. Load data dari Dexie.js ke dalam Pinia Store
      await fetchCatalog()
    } catch (err) {
      console.error('Gagal memuat katalog:', err)
    }
  })

  return {
    // State Reaktif
    products,
    categories,
    selectedCategory,
    searchQuery,
    filteredProducts,
    isLoading,

    // Actions
    setCategory,
    setSearchQuery,
    fetchCatalog
  }
}
