===== src/stores/catalogStore.js =====
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { productService } from '../services/productService'

export const useCatalogStore = defineStore('catalog', () => {
  const products = ref([])
  const categories = ref([])
  const selectedCategory = ref('SEMUA')
  const searchQuery = ref('')
  const isLoading = ref(false)

  async function fetchCatalog() {
    isLoading.value = true
    try {
      products.value = await productService.getAllProducts()
      categories.value = await productService.getCategories()
      // Sort kategori by jumlah produk terbanyak
      categories.value.sort((a,b) => b.count - a.count)
    } catch (e) {
      console.error(e)
    } finally {
      isLoading.value = false
    }
  }

  const filteredProducts = computed(() => {
    return products.value.filter(p => {
      const matchCat = selectedCategory.value === 'SEMUA' || p.category === selectedCategory.value
      const matchSearch = p.name.toLowerCase().includes(searchQuery.value.toLowerCase().trim())
      return matchCat && matchSearch && p.isAvailable
    })
  })

  function setCategory(name) { selectedCategory.value = name }
  function setSearchQuery(q) { searchQuery.value = q }

  return { products, categories, selectedCategory, searchQuery, isLoading, filteredProducts, fetchCatalog, setCategory, setSearchQuery }
})

===== src/stores/cartStore.js =====
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { whatsappService } from '../services/whatsappService'

export const useCartStore = defineStore('cart', () => {
  const items = ref([])

  const totalItems = computed(() => items.value.reduce((sum, item) => sum + item.quantity, 0))
  const totalPrice = computed(() => items.value.reduce((sum, item) => sum + (item.price * item.quantity), 0))

  function addItem(product, variant = null) {
    const existing = items.value.find(i => i.id === product.id && i.selectedVariant === variant)
    if (existing) {
      existing.quantity++
    } else {
      items.value.push({
        id: product.id,
        name: product.name,
        price: product.price,
        unit: product.unit,
        selectedVariant: variant,
        quantity: 1
      })
    }
  }

  function removeItem(productId) {
    items.value = items.value.filter(i => i.id !== productId)
  }

  function updateQuantity(productId, qty) {
    const item = items.value.find(i => i.id === productId)
    if (item) {
      item.quantity = qty
      if (item.quantity <= 0) removeItem(productId)
    }
  }

  function checkoutViaWhatsApp(storePhone, customerInfo) {
    const message = whatsappService.formatOrderMessage(items.value, customerInfo)
    const url = whatsappService.generateWhatsAppURL(storePhone, message)
    window.open(url, '_blank')
  }

  return { items, totalItems, totalPrice, addItem, removeItem, updateQuantity, checkoutViaWhatsApp }
})

import fullDB from '../mock/full-db.json' // ganti nama file kamu yang 474 produk jadi full-db.json

export function useCatalog() {
  const catalogStore = useCatalogStore()
  const { categories, selectedCategory, searchQuery, filteredProducts, isLoading } = storeToRefs(catalogStore)
  const { setCategory, setSearchQuery, fetchCatalog } = catalogStore

  onMounted(async () => {
    try {
      await productService.initFromFullDB(fullDB)
      await fetchCatalog()
    } catch (err) {
      console.error('Gagal memuat katalog:', err)
    }
  })

  return { categories, selectedCategory, searchQuery, filteredProducts, isLoading, setCategory, setSearchQuery, fetchCatalog }
}

import { db } from '../db/index'

// Mapping kategori database real -> icon Lucide (21 kategori)
export const CATEGORY_ICON_MAP = {
  "POP MIE": "UtensilsCrossed",
  "MIE": "UtensilsCrossed",
  "MIE GELAS": "CupSoda",
  "MIE GAUL": "Flame",
  "MINUMAN SACHET": "Coffee",
  "MINUMAN BOTOL": "CupSoda",
  "MINUMAN CUP": "CupSoda",
  "POP ICE": "Snowflake",
  "ES CEKEK": "Snowflake",
  "SNACK 1K": "Cookie",
  "SNACK 2K": "Cookie",
  "SNACK GOPE": "Cookie",
  "PERMEN": "Candy",
  "HOME CARE": "Sparkles",
  "PERSONAL CARE": "Heart",
  "BUMBU": "Soup",
  "ROKOK": "Cigarette",
  "OBAT": "Pill",
  "PEMBALUT": "Heart",
  "BARTIP": "Wrench",
  "UMUM": "Package"
}

export const productService = {
  transformPOSProduct(rawProduct, allServices = []) {
    const activeServices = allServices.filter(s => !s.deleted)
    
    // Cari service yang relevan dengan kategori produk
    const availableServices = activeServices.filter(s => {
      return s.category === rawProduct.category || 
             (rawProduct.category.includes('MIE') && s.category.includes('MIE')) ||
             (rawProduct.category === 'POP ICE' && s.name.includes('Pop Ice')) ||
             (rawProduct.category.includes('MINUMAN') && s.category === 'MINUMAN SACHET')
    })

    return {
      id: rawProduct.id,
      name: rawProduct.name,
      code: rawProduct.code || '',
      category: rawProduct.category || 'UMUM',
      categoryIcon: CATEGORY_ICON_MAP[rawProduct.category] || 'Package',
      price: rawProduct.price_sell || 0,
      unit: rawProduct.unit || 'PCS',
      stock: rawProduct.qty || 0,
      minStock: rawProduct.minStock || 2,
      isLowStock: (rawProduct.qty || 0) <= (rawProduct.minStock || 2) && (rawProduct.qty || 0) > 0,
      isAvailable: (rawProduct.qty || 0) > 0,
      availableServices,
      // Placeholder dulu, nanti ganti ke Cloudinary kalau sudah ada foto
      imageUrl: `https://placehold.co/300x300/e2e8f0/475569?text=${encodeURIComponent(rawProduct.name.substring(0,12))}`,
      // Untuk nanti Cloudinary tinggal ganti baris di atas jadi:
      // imageUrl: `https://res.cloudinary.com/YOUR_CLOUD_NAME/image/upload/f_auto,q_auto,w_300/v1/products/${rawProduct.id}.jpg`,
      updatedAt: rawProduct.updatedAt
    }
  },

  async initFromFullDB(fullDB) {
    const count = await db.products.count()
    if (count > 0) return // sudah ada

    const activeServices = (fullDB.services || []).filter(s => !s.deleted)
    await db.services.bulkAdd(activeServices)

    const transformed = (fullDB.products || []).map(p => this.transformPOSProduct(p, activeServices))
    await db.products.bulkAdd(transformed)

    // Build 21 kategori real dari database
    const catMap = {}
    transformed.forEach(p => {
      if (!catMap[p.category]) {
        catMap[p.category] = {
          id: p.category,
          name: p.category,
          icon: p.categoryIcon,
          count: 0
        }
      }
      catMap[p.category].count++
    })
    await db.categories.bulkAdd(Object.values(catMap))

    if (fullDB.settings?.[0]) {
      await db.settings.put({ id: 'app_config', ...fullDB.settings[0] })
    }
  },

  async getAllProducts() { return await db.products.toArray() },
  async getCategories() { return await db.categories.toArray() },
  async getSettings() { return await db.settings.get('app_config') }
}

===== src/services/whatsappService.js =====
export const whatsappService = {
  formatOrderMessage(cartItems, customerInfo = {}) {
    const { name = 'Anak Kost', room = '-' } = customerInfo
    
    let text = `*PESANAN BARU - WARUNG SINAR PAGI*\n`
    text += `-----------------------------------\n`
    text += `*Pemesan:* ${name}\n`
    text += `*No. Kost/Kamar:* ${room}\n\n`
    text += `*Rincian Belanjaan:*\n`

    let total = 0
    cartItems.forEach((item, index) => {
      const subtotal = item.price * item.quantity
      total += subtotal
      text += `${index + 1}. ${item.name} (${item.quantity} ${item.unit}) = Rp ${subtotal.toLocaleString('id-ID')}\n`
      if (item.selectedVariant) {
        text += `   _Catatan: ${item.selectedVariant}_\n`
      }
    })

    text += `\n-----------------------------------\n`
    text += `*TOTAL BAYAR:* Rp ${total.toLocaleString('id-ID')}\n\n`
    text += `Mohon konfirmasi pesanan dan info nomor QRIS / COD ya A/Teteh. Terima kasih!`

    return encodeURIComponent(text)
  },

  generateWhatsAppURL(phoneNumber, formattedMessage) {
    const cleanPhone = phoneNumber.replace(/[^0-9]/g, '')
    return `https://wa.me/${cleanPhone}?text=${formattedMessage}`
  }
}
